    <script>
        const prizes = [
            { id: 1, name: "頭獎", winner: null },
            { id: 2, name: "二獎", winner: null },
            { id: 3, name: "三獎", winner: null }
        ];

        let currentPrizeIndex = 0;
        let pool = [];
        let isDrawing = false;

        // Initialize UI
        window.onload = function() {
            updateParticipants();
            renderPrizes();
        };

        function updateParticipants() {
            const rawText = document.getElementById('participantsInput').value;
            // Split lines, trim, an"></div>
                <div class="title-text">
                    <h1>阿呆AI 幸運抽獎</h1>
                    <p>貼上參與者名單，依序抽出頭獎、二獎與三獎，並區出中獎名單。</p>
                </div>
            </div>
            <div class="progress-pill" id="progressPill">
                已抽出 0/3 個獎項
            </div>
        </header>

        <!-- Main Content Grid -->
        <div class="grid-layout">
            <!-- Draw Box -->
            <div class="card draw-stage"d remove empty + duplicates
            const lines = rawText.split('\n')
                .map(line => line.trim())
                .filter(line => line.length > 0);
            
            // Remove duplicates
            const uniqueLines = Array.from(new Set(lines));

            // Exclude already won participants
            const winners = prizes.map(p => p.winner).filter(Boolean);
            pool = uniqueLines.filter(p => !winners.includes(p));

            document.getElementById('participantCountBadge').innerText = `${uniqueLines.length} 人`;
            document.getElementById('remainingCountBadge').innerText = `👥 剩餘 ${pool.length} 人`;
        }

        function renderPrizes() {
            const container = document.getElementById('prizeList');
            container.innerHTML = '';

            prizes.forEach((prize, index) => {
                const item = document.createElement('div');
                item.className = `prize-item ${index === currentPrizeIndex && !prize.winner ? 'active' : ''}`;
                
                let statusText = "待抽出 (按「開始抽獎」)";
                if (prize.winner) {
                    statusText = `🎉 得主：${prize.winner}`;
                }

                item.innerHTML = `
                    <div class="prize-header">
                        <div class="prize-number">${prize.id}</div>
                        <div class="prize-name">${prize.name}</div>
                    </div>
                    <div class="prize-status">${statusText}</div>
                `;
                container.appendChild(item);
            });

            // Update header progress
            const drawnCount = prizes.filter(p => p.winner !== null).length;
            document.getElementById('progressPill').innerText = `已抽出 ${drawnCount}/${prizes.length} 個獎項`;

            // Update current prize badge
            const currentPrize = prizes[currentPrizeIndex];
            if (currentPrize && !currentPrize.winner) {
                document.getElementById('currentPrizeBadge').innerText = `✨ 正在抽出：${currentPrize.name}`;
                document.getElementById('drawBtn').disabled = false;
            } else if (drawnCount === prizes.length) {
                document.getElementById('currentPrizeBadge').innerText = `🎉 抽獎完畢`;
                document.getElementById('drawBtn').disabled = true;
            }
        }

        function startDraw() {
            if (isDrawing) return;
            if (currentPrizeIndex >= prizes.length) {
                alert('所有獎項已抽完！');
                return;
            }
            if (pool.length === 0) {
                alert('請先在右側輸入參與者名單！');
                return;
            }

            isDrawing = true;
            const drawBtn = document.getElementById('drawBtn');
            const displayArea = document.getElementById('displayArea');
            drawBtn.disabled = true;

            let counter = 0;
            const duration = 2000; // 2 seconds animation
            const intervalTime = 50;
            const totalSteps = duration / intervalTime;

            const timer = setInterval(() => {
                const randomName = pool[Math.floor(Math.random() * pool.length)];
                displayArea.innerText = randomName;
                counter++;

                if (counter >= totalSteps) {
                    clearInterval(timer);
                    // Final winner select
                    const winnerIndex = Math.floor(Math.random() * pool.length);
                    const winner = pool[winnerIndex];
                    
                    prizes[currentPrizeIndex].winner = winner;
                    displayArea.innerText = winner;

                    // Update Lists
                    renderWinners();
                    
                    // Move to next prize
                    currentPrizeIndex++;
                    updateParticipants();
                    renderPrizes();

                    isDrawing = false;
                }
            }, intervalTime);
        }

        function renderWinners() {
            const winners = prizes.filter(p => p.winner !== null);
            const container = document.getElementById('winnersContainer');
            const emptyText = document.getElementById('emptyWinnersText');

            if (winners.length > 0 && emptyText) {
                emptyText.remove();
            }

            container.innerHTML = '';
            winners.forEach(p => {
                const item = document.createElement('div');
                item.className = 'winner-item';
                item.innerHTML = `
                    <span class="winner-prize">${p.name}</span>
                    <span>${p.winner}</span>
                `;
                container.appendChild(item);
            });
        }
